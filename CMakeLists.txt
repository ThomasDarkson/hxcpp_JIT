cmake_minimum_required(VERSION 3.16)
project(hxcpp_JIT C CXX)
option(HXCPP_JIT_MINGW "Build the LLVM-MinGW JIT backend (Windows default)" ${WIN32})
if(HXCPP_JIT_MINGW)
  if(NOT MINGW OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "HXCPP_JIT_MINGW requires the x64 LLVM-MinGW Clang compiler.")
  endif()
elseif(MINGW)
  message(FATAL_ERROR "Select -DHXCPP_JIT_MINGW=ON for a MinGW build.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "hxcpp_JIT supports Windows and Linux x86_64.")
endif()
if(NOT WIN32 AND (NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$"))
  message(FATAL_ERROR "The Linux backend requires x86_64.")
endif()

find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)

get_filename_component(_llvm_bin "${LLVM_TOOLS_BINARY_DIR}" ABSOLUTE)
set(CLANG_RESOURCE_DIR "" CACHE PATH "Clang resource directory")
foreach(_exe "${_llvm_bin}/clang" "${_llvm_bin}/clang.exe")
  if(EXISTS "${_exe}" AND CLANG_RESOURCE_DIR STREQUAL "")
    execute_process(COMMAND "${_exe}" -print-resource-dir
                    OUTPUT_VARIABLE CLANG_RESOURCE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  endif()
endforeach()
file(TO_CMAKE_PATH "${CLANG_RESOURCE_DIR}" CLANG_RESOURCE_DIR)

if(HXCPP_JIT_MINGW)
  if(NOT LLVM_HOST_TRIPLE MATCHES "(mingw|windows-gnu)")
    message(FATAL_ERROR "LLVM must also be built with LLVM-MinGW; MSVC LLVM libraries cannot be reused.")
  endif()
  add_library(hxcpp_JIT SHARED native/hxcpp_JIT.cpp native/mingw_runtime.cpp native/hxcpp_JIT.def)
  target_compile_definitions(hxcpp_JIT PRIVATE HXCPP_JIT_MINGW=1)
  set_target_properties(hxcpp_JIT PROPERTIES PREFIX "")
  # Only the public C API is exported by the .def file.
  target_link_options(hxcpp_JIT PRIVATE "-Wl,--exclude-all-symbols")
elseif(WIN32)
  add_library(hxcpp_JIT SHARED native/hxcpp_JIT.cpp native/msvc_runtime.cpp)
else()
  add_library(hxcpp_JIT SHARED native/hxcpp_JIT.cpp)
  set(HXCPP_JIT_COMPILER_RT_SOURCE "" CACHE PATH "Matching compiler-rt source directory")
  if(NOT EXISTS "${HXCPP_JIT_COMPILER_RT_SOURCE}/lib/builtins/emutls.c")
    message(FATAL_ERROR "Set HXCPP_JIT_COMPILER_RT_SOURCE to the matching compiler-rt source; scripts/build-linux.sh does this automatically.")
  endif()
  target_sources(hxcpp_JIT PRIVATE "${HXCPP_JIT_COMPILER_RT_SOURCE}/lib/builtins/emutls.c")
  set_target_properties(hxcpp_JIT PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
target_include_directories(hxcpp_JIT PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS} native)
target_compile_definitions(hxcpp_JIT PRIVATE NOMINMAX ${LLVM_DEFINITIONS})
if(NOT CLANG_RESOURCE_DIR STREQUAL "")
  target_compile_definitions(hxcpp_JIT PRIVATE HXCPP_JIT_CLANG_RESOURCE_DIR="${CLANG_RESOURCE_DIR}")
endif()

if(MSVC)
  set_source_files_properties(native/hxcpp_JIT.cpp PROPERTIES COMPILE_OPTIONS "/GR-")
else()
  set_source_files_properties(native/hxcpp_JIT.cpp PROPERTIES COMPILE_OPTIONS "-fno-rtti")
endif()

llvm_map_components_to_libnames(_llvm_libs core support orcjit native bitreader bitwriter targetparser)
target_link_libraries(hxcpp_JIT PRIVATE clangInterpreter clangFrontend ${_llvm_libs})
if(WIN32 AND NOT HXCPP_JIT_MINGW)
  set_target_properties(hxcpp_JIT PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
target_link_libraries(hxcpp_JIT PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(hxcpp_JIT PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Threads REQUIRED)
  target_link_libraries(hxcpp_JIT PRIVATE Threads::Threads)
  option(HXCPP_JIT_STATIC_CXX_RUNTIME "Include the complete GCC C++ runtime" ON)
  if(HXCPP_JIT_STATIC_CXX_RUNTIME)
    foreach(_archive libstdc++.a libgcc.a)
      execute_process(COMMAND "${CMAKE_CXX_COMPILER}" "-print-file-name=${_archive}"
        OUTPUT_VARIABLE _runtime OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _status)
      if(NOT _status EQUAL 0 OR NOT IS_ABSOLUTE "${_runtime}" OR NOT EXISTS "${_runtime}")
        message(FATAL_ERROR "Missing static runtime: ${_archive}")
      endif()
      target_link_libraries(hxcpp_JIT PRIVATE "-Wl,--whole-archive" "${_runtime}" "-Wl,--no-whole-archive")
      if(_archive STREQUAL "libstdc++.a")
        set(_libstdcxx_archive "${_runtime}")
      endif()
    endforeach()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(_runtime_exports "${CMAKE_CURRENT_BINARY_DIR}/linux-exports.map")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/linux/runtime-exports.py"
      "${CMAKE_NM}" "${_libstdcxx_archive}" "${_runtime_exports}" RESULT_VARIABLE _status)
    if(NOT _status EQUAL 0)
      message(FATAL_ERROR "Cannot generate Linux runtime exports")
    endif()
    target_link_options(hxcpp_JIT PRIVATE "-Wl,--version-script=${_runtime_exports}" -Wl,--gc-sections -static-libstdc++ -shared-libgcc)
    set_property(TARGET hxcpp_JIT APPEND PROPERTY LINK_DEPENDS "${_runtime_exports}")
  endif()
endif()
